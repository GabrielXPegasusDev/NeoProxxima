CREATE TABLE Plano (
    ID_Plano INT AUTO_INCREMENT,
    Nome_Plano VARCHAR(50) NOT NULL,
    Velocidade_Mbps INT NOT NULL,
    Valor_Mensal DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (ID_Plano)
) ENGINE=InnoDB;


CREATE TABLE Local_Estacoes (
    ID_Estacao INT AUTO_INCREMENT,
    Nome_Estacao VARCHAR(100) NOT NULL,
    Coordenadas_GPS VARCHAR(50) NOT NULL,
    Status_Operacional BOOLEAN DEFAULT TRUE,
    PRIMARY KEY (ID_Estacao)
) ENGINE=InnoDB;


CREATE TABLE Cliente (
    ID_Cliente INT AUTO_INCREMENT,
    Nome_Completo VARCHAR(100) NOT NULL,
    CPF VARCHAR(14) NOT NULL UNIQUE,
    Email VARCHAR(100) NOT NULL UNIQUE,
    Senha_Hash VARCHAR(255) NOT NULL,
    ID_Plano INT,
    PRIMARY KEY (ID_Cliente),
    FOREIGN KEY (ID_Plano) REFERENCES Plano(ID_Plano) 
        ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB;


CREATE TABLE Pagamentos (
    ID_Pagamento INT AUTO_INCREMENT,
    ID_Cliente INT NOT NULL,
    Data_Vencimento DATE NOT NULL,
    Status_Pagamento VARCHAR(20) NOT NULL DEFAULT 'Pendente',
    Metodo_Pagamento VARCHAR(50) NOT NULL,
    PRIMARY KEY (ID_Pagamento),
    FOREIGN KEY (ID_Cliente) REFERENCES Cliente(ID_Cliente) 
        ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;


CREATE TABLE Dispositivo (
    ID_Dispositivo INT AUTO_INCREMENT,
    ID_Cliente INT NOT NULL,
    Endereco_MAC CHAR(17) NOT NULL,
    Tipo_Dispositivo VARCHAR(30) NOT NULL,
    PRIMARY KEY (ID_Dispositivo),
    FOREIGN KEY (ID_Cliente) REFERENCES Cliente(ID_Cliente) 
        ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;


CREATE TABLE Registro_Conexao (
    ID_Conexao INT AUTO_INCREMENT,
    ID_Cliente INT NOT NULL,
    ID_Estacao INT NOT NULL,
    Data_Hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    Consumo_MB DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (ID_Conexao),
    FOREIGN KEY (ID_Cliente) REFERENCES Cliente(ID_Cliente) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (ID_Estacao) REFERENCES Local_Estacoes(ID_Estacao) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;


CREATE TABLE Chamada_Suporte (
    ID_Chamada INT AUTO_INCREMENT,
    ID_Cliente INT NOT NULL,
    ID_Estacao INT NOT NULL,
    Problema_Desc TEXT NOT NULL,
    Status BOOLEAN DEFAULT TRUE,
    PRIMARY KEY (ID_Chamada),
    FOREIGN KEY (ID_Cliente) REFERENCES Cliente(ID_Cliente) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (ID_Estacao) REFERENCES Local_Estacoes(ID_Estacao) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;
